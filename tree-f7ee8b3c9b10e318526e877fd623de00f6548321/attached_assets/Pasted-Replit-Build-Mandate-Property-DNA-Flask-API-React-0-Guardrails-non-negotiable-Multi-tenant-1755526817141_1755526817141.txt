DNA generator (/packages/utils/dna.py):

import base64, hashlib, datetime
def base32_6(b: bytes) -> str:
    return base64.b32encode(b)[:6].decode("ascii").rstrip("=")
def make_dna_id(country, muni, parcel, unit, deed):
    seed = f"{country}|{muni}|{parcel}|{unit or ''}|{deed or ''}".encode()
    rnd = base32_6(hashlib.sha256(seed).digest())
    year = datetime.date.today().year
    parts = ["PDNA", country, muni, parcel] + ([unit] if unit else [])
    return "-".join(parts + [str(year), rnd])


S3/R2 signed PUT (/apps/api/src/routes/documents.py):

import boto3, hashlib, time, os
from flask import Blueprint, request, jsonify
bp = Blueprint("documents", __name__)
s3 = boto3.client("s3",
    endpoint_url=os.environ["S3_ENDPOINT"],
    aws_access_key_id=os.environ["S3_ACCESS_KEY_ID"],
    aws_secret_access_key=os.environ["S3_SECRET_ACCESS_KEY"],
)
BUCKET = os.environ["S3_BUCKET"]

@bp.post("/api/documents/sign")
def sign_upload():
    j = request.get_json()
    key = j["key"]               # e.g. f"{org_id}/{property_id}/{ts}-{filename}"
    content_type = j["contentType"]
    sha256 = j["sha256"]
    bytes_len = int(j["bytes"])
    # presigned PUT
    url = s3.generate_presigned_url(
        ClientMethod='put_object',
        Params={"Bucket": BUCKET, "Key": key, "ContentType": content_type},
        ExpiresIn=900, HttpMethod="PUT"
    )
    return jsonify({"url": url, "storageKey": key, "contentType": content_type, "sha256": sha256, "bytes": bytes_len})


Audit helper (/apps/api/src/audit.py):

from functools import wraps
from flask import g
from .models import AuditLog, db
def audit(action, target_type):
    def deco(f):
        @wraps(f)
        def inner(*args, **kwargs):
            resp = f(*args, **kwargs)
            db.session.add(AuditLog(org_id=g.org_id, actor_user_id=getattr(g,"user_id",None),
                                    action=action, target_type=target_type,
                                    target_id=kwargs.get("id"), payload={}))
            db.session.commit()
            return resp
        return inner
    return deco
